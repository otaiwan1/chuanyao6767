# sixty_seven.py — 數一數有幾個 67
def count_67(nums):
    """回傳串列中 67 出現次數，O(n)。"""
    return sum(1 for n in nums if n == 67)

if __name__ == "__main__":
    data = [6, 7, 67, 6767, 67]
    print(f"Hello, 電神! 67 出現 {count_67(data)} 次")
// sixtySeven.js — 本站 67 計數器核心邏輯
function fib67(n) {
  let a = 0, b = 1;
  for (let i = 0; i < n; i++) [a, b] = [b, a + b];
  return a;
}
console.log(`Hello, 電神! fib(6)=${fib67(6)}, fib(7)=${fib67(7)}`);
